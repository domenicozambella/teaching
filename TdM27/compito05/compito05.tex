\begin{exercise}(Martina Apicella \& Andrea Morello)
  Let $\phi(x\,;z)\in L(A)$. 
  Let $M\supseteq A$ be a sufficiently saturated model.
  Let say that $p(x)\in S_\phi(M)$ be invariant over $A$ if \ssf{it2$^\prime$.} holds for every $a,b\in M^z$.
  Prove that there is the unique global $\phi$-type that extends $p(x)$ and is invariant over $A$.
  Prove that such a type does not depend on the choice of $M$.
  What is the required degree of saturation?
\end{exercise}


\begin{exercise}(Beatrice Calderaro)
  Let $N\succeq M$ be saturated and of cardinality $>|M|$.
  Let $a_i\nonfork_M N, b$, for $i=1,2$. 
  Prove that if $a_1\equiv_N a_2$ then $a_1\equiv_{N,b} a_2$.
\end{exercise}


\begin{exercise}(Mattia Lo Mele \& Daniele Lanciotti)
  Show that in general there is no type $p(z)\subseteq L(M,a)$ such that 

   $$a\nonfork_M b\ \ \IFF\ \  b\models p(z).$$

  Hint: consider the theory of dense linear orders without endpoints.
\end{exercise}

\begin{exercise}(Francesco Dam)
  Prove that the following are equivalent
  \begin{itemize}
  \item[1.] $T$ has weak elimination of imaginaries
  \item[2.] $\dcl^\eq A\,\cap\,\dcl^\eq B=\dcl^\eq(A\cap B)$ for every algebraically closed sets $A,B\subseteq\U$.
\end{itemize}
\end{exercise}

\begin{exercise}(Filippo Giugie \& Daniele Lanciotti)
  Let $a\nonfork_M b$. 
  Prove that for every $c$ there is $b'\equiv_{M,a} b$ such that $a,c\nonfork_M b'$. 
\end{exercise}
