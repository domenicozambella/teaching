\begin{exercise}
  (Alessandro Martina)
  Show that in general there is no type $p(x\,;z)\subseteq L(M)$ such that for every  $a\in\U^x$ and $b\in\U^z$\vspace*{-1ex}

  \hfil$a\,;b\models p(x\,;z)\ \ \ \IFF\ \ \ a\cnonfork_M b.$\vspace*{-1ex}

  Hint: consider the theory of dense linear orders.
\end{exercise}
\begin{exercise} 
  (Alessandro Martina)
  Let $M$ be an arbitrary model.
  Let $\phi(x,y)\in L(M)$, where $|x|=|y|$.
  Prove that there is a sequence $\langle a_i:i<\omega\rangle$ in $M^{x}$ such that\vspace*{-1ex}

  \hfil$\phi(a_i,a_j)\iff\phi(a_h,a_k)$\vspace*{-1ex}

  for every $i<j<\omega$ and $h<k<\omega$.
\end{exercise}

\begin{exercise} 
  (Francesco Sulpizi)
  Let $\bar c=\langle c_i:i\in I\rangle$ be an $A$-indiscernible sequence.
  Let $a\models p(x)$, where $p(x)\in S(A,\bar c)$ is finitely satisfiable in every $M\supseteq A$.
  Prove that $\bar c$ is indiscernible over $A,a$.
\end{exercise}

\begin{exercise}
  (Leonardo Centazzo)
  Let $\phi(x,y)\in  L$, where $|x|=|y|=1$.
  Suppose there is an infinite set $A\subseteq\U$ such that $\phi(a,b)\niff\phi(b,a)$ for every two distinct $a,b\in A$.
  Prove that $\phi(x\,;y)$ is unstable.
\end{exercise}
\begin{exercise}
  (Leonardo Centazzo)
  Let $N\succeq M$ be $\lambda$-saturated for $\lambda>|M|$.
  Let $a_i\cnonfork_M N,b$, for $i=1,2$.
  Prove that if $a_1\equiv_N a_2$ then $a_1\equiv_{N,b} a_2$.
\end{exercise}

\begin{exercise}
  (Roberto Carnevale)
  Let $M$ be a graph with the property that for every finite $A\subseteq M$ there is a $c\in M$ such that $A\subseteq r(c,M)$.
  The edges of $M$ are colored with infinitely many colors but every clique contains at most $k$ colors, for some fixed $k\in\omega$.
  Prove that $M$ has an infinite monochromatic complete subgraph.
\end{exercise}

\begin{exercise}
  (Davide Peccioli)
  Assume $L$ is countable and let $T$ be strongly minimal.
  Prove that the following are equivalent
  \begin{itemize}
  \item[1.] $T$ is $\omega$-categorical
  \item[2.] the algebraic closure of a finite set is finite.
  \end{itemize}
\end{exercise}
